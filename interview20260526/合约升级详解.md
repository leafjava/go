# 合约升级（Contract Upgrade）详解

## 一、为什么需要合约升级

区块链上智能合约一旦部署就**不可篡改**，这是区块链的核心价值——代码即法律。但这也带来了实际问题：

- **Bug 修复**：审计再严格也难保零漏洞，Parity 多签钱包一个 bug 锁死了 30 万 ETH
- **功能迭代**：业务需求会变，DeFi 协议需要加新池子、新策略
- **Gas 优化**：EVM 升级后（如 EIP-1559、Dencun），老合约无法利用新特性降 Gas
- **安全响应**：发现漏洞后需要立即修复，不能等"下次部署"

如果没有升级机制，bug 修复只能靠硬分叉或者部署新合约让用户迁移——前者几乎不可能，后者用户体验极差。

## 二、核心机制：代理模式（Proxy Pattern）

### 2.1 delegatecall 原理

普通合约调用 `call` 和代理调用的 `delegatecall` 有本质区别：

```
call:                        delegatecall:
  ┌──────┐                    ┌──────┐
  │  A   │ ─── call ───→     │  A   │ ─── delegatecall ───→
  │      │  用 B 的存储        │      │  用 A 的存储！
  └──────┘  改 B 的状态        └──────┘  改 A 的状态！
     ↑                            ↑
  调用者                        调用者
```

**一句话**：delegatecall 让逻辑合约用代理合约的存储来执行代码——代码在别处，数据留原地。

### 2.2 最小代理合约（Go + Solidity 示例）

```go
package main

import (
    "crypto/sha256"
    "encoding/hex"
    "fmt"
    "math/big"
)

// ─── Go 端：部署并管理代理合约 ───

// ProxyDeployer 代理合约部署器
type ProxyDeployer struct {
    rpcClient *RPCClient
}

// DeployMinimalProxy 部署最小代理（EIP-1167 Minimal Proxy）
//
// EIP-1167 最小代理只需要 ~50 字节的 bytecode，
// 将所有调用通过 delegatecall 转发到固定的逻辑合约地址。
// 相比完整 Proxy 合约，部署 Gas 从 ~200K 降到 ~50K。
func (d *ProxyDeployer) DeployMinimalProxy(logicContract string) (string, error) {
    // EIP-1167 最小代理的 bytecode
    // 核心逻辑只有一段：把 calldata 原封不动 delegatecall 到逻辑合约
    //
    // 伪代码等价于:
    //   assembly {
    //       calldatacopy(0, 0, calldatasize())
    //       result := delegatecall(gas(), implementation, 0, calldatasize(), 0, 0)
    //       returndatacopy(0, 0, returndatasize())
    //       switch result
    //       case 0 { revert(0, returndatasize()) }
    //       default { return(0, returndatasize()) }
    //   }

    proxyBytecode := buildEIP1167Bytecode(logicContract)

    // 部署代理合约
    proxyAddr, err := d.rpcClient.DeployContract(proxyBytecode)
    if err != nil {
        return "", fmt.Errorf("部署代理失败: %w", err)
    }

    fmt.Printf("最小代理已部署: %s → 逻辑合约: %s\n", proxyAddr, logicContract)
    fmt.Printf("  部署 Gas: ~50K（完整Proxy需 ~200K）\n")

    return proxyAddr, nil
}

// buildEIP1167Bytecode 构建 EIP-1167 最小代理 bytecode
func buildEIP1167Bytecode(logic string) []byte {
    // EIP-1167 固定模板：
    // 3d602d80600a3d3981f3 - 创建时代码（clone 逻辑）
    // 363d3d373d3d3d363d73 - 运行时代码前缀（calldata 转发）
    // <20 bytes logic address> - 逻辑合约地址
    // 5af43d82803e903d91602b57fd5bf3 - 运行时代码后缀（delegatecall + 返回）

    prefix := "363d3d373d3d3d363d73"
    suffix := "5af43d82803e903d91602b57fd5bf3"

    // 去掉 0x 前缀，拼接
    addr := logic[2:] // 假设 logic 是 "0x..." 格式
    bytecodeHex := "3d602d80600a3d3981f3" + prefix + addr + suffix

    bytecode, _ := hex.DecodeString(bytecodeHex)
    return bytecode
}
```

**对应的 Solidity 代码**：

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

// ─── 最简代理（教学用，不要直接用于生产）───

contract MinimalProxy {
    // 逻辑合约地址，存在 proxy 的 storage slot 0
    address public implementation;

    constructor(address _implementation) {
        implementation = _implementation;
    }

    // 核心：所有调用通过 delegatecall 转发给逻辑合约
    // msg.sender 和 msg.value 原封不动传递
    fallback() external payable {
        address impl = implementation;
        assembly {
            // 复制 calldata 到内存
            calldatacopy(0, 0, calldatasize())

            // delegatecall 到逻辑合约
            // 参数: gas, address, argsOffset, argsSize, retOffset, retSize
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)

            // 复制返回数据
            returndatacopy(0, 0, returndatasize())

            // 根据 delegatecall 结果决定 revert 还是 return
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }
}
```

## 三、三种主流代理方案

### 3.1 透明代理（Transparent Proxy）

**核心特点**：升级函数在 Proxy 合约端，管理员调用不会被 delegatecall 转发。

```
调用流程：

  普通用户 → Proxy → delegatecall → Logic（执行逻辑）
  管理员   → Proxy → 直接处理（upgradeTo），不转发
```

```solidity
// ─── Transparent Proxy（OpenZeppelin 标准实现简化版）───

contract TransparentProxy {
    // slot 0: 逻辑合约地址
    address private _implementation;
    // slot 1: 管理员地址（存在特定 slot 避免存储冲突）
    address private _admin;

    event Upgraded(address indexed implementation);

    // 只有管理员能调用的修饰符
    modifier ifAdmin() {
        if (msg.sender == _admin) {
            _; // 管理员调用 → 直接在 Proxy 处理，不转发
        } else {
            _fallback(); // 普通用户 → delegatecall 到逻辑合约
        }
    }

    // 管理员直接调用的函数（只在 Proxy 内部执行）
    function upgradeTo(address newImplementation) external ifAdmin {
        _implementation = newImplementation;
        emit Upgraded(newImplementation);
    }

    function admin() external ifAdmin returns (address) {
        return _admin;
    }

    // 所有其他调用 → 转发给逻辑合约
    fallback() external payable {
        _fallback();
    }

    receive() external payable {
        _fallback();
    }

    function _fallback() internal {
        address impl = _implementation;
        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }
}

// ─── Transparent Proxy 的逻辑合约 ───
contract MyLogicV1 {
    uint256 public value;

    // 不能用 constructor，用 initialize 替代
    function initialize(uint256 _value) external {
        require(value == 0, "already initialized");
        value = _value;
    }

    function setValue(uint256 _newValue) external {
        value = _newValue;
    }
}
```

**优点**：升级逻辑在 Proxy 端，逻辑合约出错不会影响升级能力
**缺点**：每次调用都检查 `msg.sender == admin`，Gas 略高

### 3.2 UUPS（通用可升级代理标准，EIP-1822）

**核心特点**：升级函数在逻辑合约端，通过 delegatecall 执行，更省 Gas。

```
调用流程：

  所有用户（包括管理员）→ Proxy → delegatecall → Logic
                                    ↑
                               upgradeTo 也在 Logic 里
                               管理员调用时执行 _authorizeUpgrade 检查
```

```solidity
// ─── UUPS Proxy（OpenZeppelin 标准实现简化版）───

// UUPS Proxy 本身极其简单，只是一个 delegatecall 转发器
contract UUPSProxy {
    address public implementation;

    constructor(address _implementation) {
        implementation = _implementation;
    }

    fallback() external payable {
        address impl = implementation;
        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }
}

// UUPS 的逻辑合约 —— 升级函数写在这里
abstract contract UUPSUpgradeable {
    // slot 按 EIP-1967 规范存储，避免与逻辑合约变量冲突
    bytes32 private constant IMPLEMENTATION_SLOT =
        0x360894a13ba1a3210667c828492db98dca3e2076cc3735a920a3ca505d382bbc;

    event Upgraded(address indexed implementation);

    // 子合约必须实现这个方法来做权限检查
    function _authorizeUpgrade(address newImplementation) internal virtual;

    // 升级函数：只有通过 _authorizeUpgrade 检查的人才能调用
    function upgradeTo(address newImplementation) external {
        _authorizeUpgrade(newImplementation);
        _setImplementation(newImplementation);
        emit Upgraded(newImplementation);
    }

    function _setImplementation(address newImplementation) internal {
        assembly {
            sstore(IMPLEMENTATION_SLOT, newImplementation)
        }
    }
}

// V1 逻辑合约
contract MyContractV1 is UUPSUpgradeable {
    uint256 public value;

    function initialize(uint256 _value) external {
        require(value == 0, "already initialized");
        value = _value;
    }

    function setValue(uint256 _newValue) external {
        value = _newValue;
    }

    // ⚠️ 关键：实现权限检查
    function _authorizeUpgrade(address) internal override onlyOwner {}
    // onlyOwner 来自继承 Ownable
}

// V2 逻辑合约 —— 新增功能 + 保持存储布局
contract MyContractV2 is UUPSUpgradeable {
    // ⚠️ 变量的顺序和类型必须与 V1 完全一致！
    uint256 public value;
    // ✅ 新变量只能加在末尾
    string public version;

    function setValue(uint256 _newValue) external {
        value = _newValue;
    }

    function getVersion() external view returns (string memory) {
        return version;
    }

    function _authorizeUpgrade(address) internal override onlyOwner {}
}
```

**为什么交易所选 UUPS**：
- 省 Gas：不需要每次调用检查 `ifAdmin`
- Proxy bytecode 极小：只有 delegatecall 转发，部署便宜
- 升级逻辑可升级：`upgradeTo` 本身在逻辑合约里，可以跟随版本演进

### 3.3 信标代理（Beacon Proxy）

**核心特点**：多个 Proxy 共享一个 Beacon 合约，改一次 Beacon 全部升级。

```
               ┌──────────────┐
               │   Beacon     │ ← 存储逻辑合约地址
               │  (信标合约)   │
               └──┬───┬───┬──┘
          ┌──────┘   │   └──────┐
          ↓          ↓          ↓
    ┌─────────┐ ┌─────────┐ ┌─────────┐
    │ Proxy 1 │ │ Proxy 2 │ │ Proxy N │  ← 所有 Proxy 从 Beacon 读取逻辑地址
    └─────────┘ └─────────┘ └─────────┘
```

```solidity
// ─── Beacon Proxy ───

// 信标合约：存储并管理逻辑合约地址
contract UpgradeableBeacon {
    address private _implementation;

    event Upgraded(address indexed implementation);

    constructor(address implementation_) {
        _implementation = implementation_;
    }

    function implementation() external view returns (address) {
        return _implementation;
    }

    // 升级信标 → 所有关联的 Proxy 自动升级
    function upgradeTo(address newImplementation) external onlyOwner {
        _implementation = newImplementation;
        emit Upgraded(newImplementation);
    }
}

// Beacon Proxy：从信标读取逻辑合约地址
contract BeaconProxy {
    address private _beacon;

    constructor(address beacon) {
        _beacon = beacon;
    }

    // 每次调用时从 Beacon 读取最新的逻辑地址
    function _implementation() internal view returns (address) {
        return UpgradeableBeacon(_beacon).implementation();
    }

    fallback() external payable {
        address impl = _implementation();
        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), impl, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }
}
```

**适用场景**：一对多场景——比如交易所需要为每个用户部署独立代理钱包，升级时改一次 Beacon 就全部升级。

### 3.4 三种代理对比

| 特性 | Transparent Proxy | UUPS | Beacon Proxy |
|------|-------------------|------|--------------|
| 升级函数位置 | Proxy 合约端 | 逻辑合约端 | Beacon 合约端 |
| Gas 开销 | 较高（每次调用检查 admin） | 低 | 中（需从 Beacon 读取） |
| 部署成本 | 高 | **最低** | 中 |
| 升级灵活性 | 中 | 高（升级逻辑可升级） | **最高**（一次改全部） |
| 升级锁死风险 | 无 | **有**（V2 忘了继承则锁死） | 低 |
| 适合场景 | 简单项目 | **交易所、DeFi 协议** | 一对多（如钱包工厂） |

## 四、钻石标准（EIP-2535 Diamond Standard）

当一个合约功能越来越多，会超过以太坊 24KB 合约大小限制。钻石标准把一个合约拆成多个 Facet（切面），每个 Facet 独立升级。

```
        ┌──────────────────────────┐
        │      Diamond Proxy       │  ← 入口合约
        │                          │
        │  facetAddress[selector]  │  ← 函数选择器 → Facet 映射表
        └───┬────────┬─────────┬───┘
            │        │         │
   ┌────────↓──┐ ┌───↓────┐ ┌──↓─────────┐
   │ Facet A   │ │Facet B │ │ Facet C    │
   │ (ERC20)   │ │(质押)   │ │(治理)      │
   │ 可独立升级 │ │可独立升级│ │ 可独立升级  │
   └───────────┘ └────────┘ └────────────┘
```

```solidity
// ─── Diamond 标准简化实现 ───

contract Diamond {
    // 选择器 → Facet 地址 + 选择器列表
    struct FacetAddressAndSelectorPosition {
        address facetAddress;
        uint16 selectorPosition; // 在 facetSelectors 数组中的位置
    }

    mapping(bytes4 => FacetAddressAndSelectorPosition) private selectorToFacet;
    bytes4[] private facetSelectors; // 所有已注册的 selector

    // 核心：根据 msg.sig 找到对应的 Facet 并 delegatecall
    fallback() external payable {
        FacetAddressAndSelectorPosition storage f =
            selectorToFacet[msg.sig];
        require(f.facetAddress != address(0), "function not found");

        assembly {
            calldatacopy(0, 0, calldatasize())
            let result := delegatecall(gas(), f.facetAddress, 0, calldatasize(), 0, 0)
            returndatacopy(0, 0, returndatasize())
            switch result
            case 0 { revert(0, returndatasize()) }
            default { return(0, returndatasize()) }
        }
    }

    // 添加/替换 Facet（只有管理员能调用）
    function diamondCut(
        FacetCut[] memory _cuts,
        address _init,
        bytes memory _calldata
    ) external onlyOwner {
        for (uint i = 0; i < _cuts.length; i++) {
            FacetCut memory cut = _cuts[i];
            // 注册每个函数选择器到对应的 Facet
            for (uint j = 0; j < cut.functionSelectors.length; j++) {
                selectorToFacet[cut.functionSelectors[j]] = FacetAddressAndSelectorPosition({
                    facetAddress: cut.facetAddress,
                    selectorPosition: uint16(facetSelectors.length)
                });
                facetSelectors.push(cut.functionSelectors[j]);
            }
        }
    }

    struct FacetCut {
        address facetAddress;
        bytes4[] functionSelectors;
    }
}
```

## 五、存储布局兼容性

### 5.1 核心规则

Solidity 合约的存储是按 slot 顺序排列的。代理模式下，Proxy 的存储布局由逻辑合约的变量声明顺序决定。

```
规则 1: 不能删变量（会造成 slot 偏移）
规则 2: 不能在已有变量中间插入新变量（只能末尾追加）
规则 3: 不能改变变量的类型
规则 4: 重命名不影响 slot（slot 按顺序，和名字无关）
```

### 5.2 错误 vs 正确的版本演进

```solidity
// V1 存储布局
contract V1 {
    uint256 public a;     // slot 0
    uint256 public b;     // slot 1
    mapping(address => uint256) public c; // slot 2
}

// ❌ 错误 V2：在中间插入了 x，导致 c 的 slot 变成了 3
contract V2_WRONG {
    uint256 public a;     // slot 0 ✓
    uint256 public x;     // slot 1 ← 新插入！读到的是旧 V1 的 b 的值
    uint256 public b;     // slot 2 ← 偏移了！读到的是旧 V1 的 c 的 mapping 占位
    mapping(address => uint256) public c; // slot 3 ← 空的！映射数据全丢了
}

// ✅ 正确 V2：新变量只能加在最后
contract V2_CORRECT {
    uint256 public a;     // slot 0 ✓
    uint256 public b;     // slot 1 ✓
    mapping(address => uint256) public c; // slot 2 ✓
    // ↑ V1 的变量保持原样（顺序、类型不能变）
    // ↓ V2 新增的变量
    uint256 public x;     // slot 3 ← 安全，因为 V1 没用到 slot 3
    string public version; // slot 4 ← 追加
}
```

### 5.3 Go 端：存储布局验证工具

```go
// ─── 存储布局兼容性检查 ───

// StorageSlot 存储槽定义
type StorageSlot struct {
    Name     string `json:"name"`
    Type     string `json:"type"`
    Slot     int    `json:"slot"`
    Offset   int    `json:"offset"`
    Size     int    `json:"size"`
}

// StorageLayout 合约存储布局
type StorageLayout struct {
    ContractName string        `json:"contract_name"`
    Version      string        `json:"version"`
    Slots        []StorageSlot `json:"slots"`
}

// ValidateStorageUpgrade 验证存储升级兼容性
//
// 输入 V1 和 V2 的存储布局（可从 Solidity 编译器输出获得），
// 输出兼容性检查结果。
func ValidateStorageUpgrade(v1, v2 StorageLayout) *CompatReport {
    report := &CompatReport{
        OldVersion: v1.Version,
        NewVersion: v2.Version,
        Compatible: true,
    }

    // 构建 V1 slot 索引
    v1Slots := make(map[int]StorageSlot)
    for _, s := range v1.Slots {
        v1Slots[s.Slot] = s
    }

    // 查找 V1 用到的最大 slot
    maxV1Slot := 0
    for _, s := range v1.Slots {
        if s.Slot > maxV1Slot {
            maxV1Slot = s.Slot
        }
    }

    for _, s := range v1.Slots {
        v2Slot, exists := findSlotByName(v2.Slots, s.Name)
        if !exists {
            report.Errors = append(report.Errors, fmt.Sprintf(
                "🔴 变量丢失: '%s' (slot %d) 在 V2 中不存在 → 数据丢失",
                s.Name, s.Slot,
            ))
            report.Compatible = false
            continue
        }

        // slot 变了？
        if v2Slot.Slot != s.Slot {
            report.Errors = append(report.Errors, fmt.Sprintf(
                "🔴 Slot 偏移: '%s' slot %d → %d",
                s.Name, s.Slot, v2Slot.Slot,
            ))
            report.Compatible = false
        }

        // 类型变了？
        if v2Slot.Type != s.Type {
            report.Errors = append(report.Errors, fmt.Sprintf(
                "🔴 类型变更: '%s' %s → %s",
                s.Name, s.Type, v2Slot.Type,
            ))
            report.Compatible = false
        }
    }

    // 新变量：必须只在 maxV1Slot 之后
    for _, s := range v2.Slots {
        if _, existed := v1Slots[s.Slot]; !existed {
            if s.Slot <= maxV1Slot {
                report.Errors = append(report.Errors, fmt.Sprintf(
                    "🔴 变量插队: '%s' 在 slot %d，但 V1 已用到 slot %d（只能追加在最后）",
                    s.Name, s.Slot, maxV1Slot,
                ))
                report.Compatible = false
            } else {
                report.Warnings = append(report.Warnings, fmt.Sprintf(
                    "🟢 新增变量: '%s' → slot %d（末尾追加，安全）",
                    s.Name, s.Slot,
                ))
            }
        }
    }

    return report
}

type CompatReport struct {
    OldVersion string
    NewVersion string
    Compatible bool
    Errors     []string
    Warnings   []string
}

func findSlotByName(slots []StorageSlot, name string) (StorageSlot, bool) {
    for _, s := range slots {
        if s.Name == name {
            return s, true
        }
    }
    return StorageSlot{}, false
}

// ─── 使用示例 ───

func demoStorageCheck() {
    v1 := StorageLayout{
        Version: "1.0.0",
        Slots: []StorageSlot{
            {Name: "owner", Type: "address", Slot: 0},
            {Name: "totalSupply", Type: "uint256", Slot: 1},
            {Name: "balances", Type: "mapping(address=>uint256)", Slot: 2},
        },
    }

    // 错误 V2：中间插入了 paused 变量
    v2Wrong := StorageLayout{
        Version: "2.0.0",
        Slots: []StorageSlot{
            {Name: "owner", Type: "address", Slot: 0},
            {Name: "paused", Type: "bool", Slot: 1},   // ← 插队！
            {Name: "totalSupply", Type: "uint256", Slot: 2},
            {Name: "balances", Type: "mapping(address=>uint256)", Slot: 3},
        },
    }
```

## 六、升级权限与安全管理

### 6.1 权限控制分层

```
升级安全金字塔:

     ┌──────────────────┐
     │   不可升级         │ ← 最安全（核心金库合约）
     │   (完全去信任)     │
     ├──────────────────┤
     │   多签 + 时间锁    │ ← 推荐（大部分业务合约）
     │   + 紧急撤回按钮   │
     ├──────────────────┤
     │   仅多签           │ ← 不够（个人被钓鱼就完蛋）
     ├──────────────────┤
     │   单 EOA 密钥      │ ← 极端危险（绝不使用）
     └──────────────────┘
```

### 6.2 时间锁 + 紧急撤回机制

升级不是即时的，而是先提交到时间锁队列等待 48h。在等待期内安全团队可以随时撤回。

```
升级生命周期：

  T+0      多签提议升级 → 进入 48h 时间锁
  T+0~48h  【监控 + 可撤回窗口】← 安全团队可随时取消
  T+48h    时间锁到期 → 执行升级

紧急情况：
  1. 安全监控发现可疑升级
  2. 告警推送 → 安全委员会确认
  3. 点击"紧急撤回"按钮 → 多签签名 → 取消升级
```

```go
    Data      []byte
    ReadyTime time.Time

    Cancelled bool
}

func (t *TimelockController) GetAllPendingOps() []*ScheduledOp {
    var ops []*ScheduledOp
    for _, op := range t.PendingOps {
        if !op.Executed && !op.Cancelled {
            ops = append(ops, op)
        }
    }
    return ops
}

func (t *TimelockController) GetPendingOpsForProxy(proxy string) []*ScheduledOp {
    var ops []*ScheduledOp
    for _, op := range t.PendingOps {
        if op.Target == proxy && !op.Executed && !op.Cancelled {
            ops = append(ops, op)
        }
    }
    return ops
}

func (t *TimelockController) CancelUpgrade(id [32]byte, initiator string, sigs [][]byte) error {
    op, ok := t.PendingOps[id]
    if !ok {
        return fmt.Errorf("操作不存在")
    }
    if op.Executed {
        return fmt.Errorf("已执行，无法取消")
    }
    op.Cancelled = true
    fmt.Printf("✓ 升级 %x 已紧急撤回\n", id[:8])
    return nil
}
```

## 七、三大常见坑与防范

### 7.1 构造函数陷阱

代理模式下 Logic 合约的 `constructor` **永远不会被调用**，因为 constructor 只在部署时执行，而 Logic 合约是独立部署的。

```solidity
// ❌ 错误：用 constructor 初始化
contract WrongLogic {
    uint256 public value;
    address public owner;

    constructor(uint256 _value) {
        value = _value;    // 这个永远不会被 Proxy 调用！
        owner = msg.sender; // Proxy 的 storage 里没有 owner！
    }
}

// ✅ 正确：用 initialize() 替代
contract CorrectLogic {
    uint256 public value;
    address public owner;
    bool private _initialized;

    function initialize(uint256 _value) external {
        require(!_initialized, "already initialized");

        // ⚠️ 必须加 initializer 修饰符，防止被重复调用
        // 否则任何人都能重新初始化，把自己设成 owner
        _initialized = true;

        value = _value;
        owner = msg.sender;
    }
}
```

### 7.2 函数选择器冲突

如果 Proxy 和 Logic 合约有同名函数（相同的 4 字节选择器），会互相覆盖。

```solidity
// Transparent Proxy 通过 ifAdmin 判断来解决：
// 管理员 → 在 Proxy 端执行（包括 upgradeTo）
// 普通用户 → delegatecall 到 Logic

// UUPS 把 upgradeTo 放在 Logic 端，避免了这个问题
// 因为所有调用都走 delegatecall，不存在 Proxy 端和 Logic 端的区分
```

### 7.3 UUPS 升级路径锁死

```solidity
// V1 正常继承 UUPSUpgradeable
contract V1 is UUPSUpgradeable {
    function _authorizeUpgrade(address) internal override onlyOwner {}
}

// 🔴 V2 忘记继承 UUPSUpgradeable → 永远无法再升级！
contract V2_BAD {
    // 没有 upgradeTo、没有 _authorizeUpgrade
    // 合约锁死在这个版本
}

// ✅ V2 必须保留 UUPSUpgradeable
contract V2_GOOD is UUPSUpgradeable {
    // V1 的变量保持原样...
    uint256 public value;

    // 新变量追加...
    string public version;

    // 必须继承 _authorizeUpgrade
    function _authorizeUpgrade(address) internal override onlyOwner {}
}
```

## 八、交易所合约升级最佳实践

### 8.1 推荐方案

```
核心金库合约 → 不可升级（最简单，最安全）
质押/收益合约 → UUPS + 多签(3/5) + 48h 时间锁 + 紧急撤回
治理合约     → UUPS + DAO 投票 + 时间锁
工厂合约     → Beacon Proxy（一拖多）
```

### 8.2 升级前 Checklist

```
□ 存储布局 diff 检查（Slither / Foundry storage check）
□ 新实现已通过至少一家审计机构审计
□ 用 mainnet fork 测试升级流程
□ 时间锁已配置（至少 24h，大额合约 48h+）
□ 管理员已是多签地址（不是单 EOA）
□ 升级事件监控已就绪
□ 紧急撤回流程已演练
□ 旧实现地址已备份（可回滚到旧版本）
□ 测试网上升级成功后再上主网
```
