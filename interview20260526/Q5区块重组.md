# 区块重组（Block Reorganization / Reorg）详解

## 一、什么是区块重组

### 1.1 一句话定义

区块重组指区块链网络出现分叉后，全网最终选择了一条更长的链，而被废弃的短链上的区块及其包含的交易被"回滚"的现象。

### 1.2 发生的根本原因

区块链是一个分布式系统，网络延迟导致不同节点可能在同一高度收到不同的区块，从而产生临时分叉：

```
时间线 →

节点 A:  区块100 ← 区块101(A) ← 区块102(A) ← 区块103(A)
节点 B:  区块100 ← 区块101(B) ← 区块102(B)
                              ↑
                         短链被废弃（重组）
                         区块101(B)、102(B) 中的交易回滚
```

### 1.3 重组的影响

- **交易回滚**：已被"确认"的交易可能从链上消失
- **双花风险**：攻击者可以利用重组实现双花攻击
- **交易所致命伤**：充值交易回滚后，交易所可能已给用户入账，造成直接资产损失

## 二、PoW 链的区块重组

### 2.1 概率性最终确定

PoW 链（BTC、旧 ETH、BSC）没有"最终确定"的概念——任何交易在数学上都可能被重组，只是概率随确认数增加而指数级下降。

```
重组概率 ≈ f(确认数, 攻击者算力占比)

以 Bitcoin 为例：
  1 确认：攻击者有 10% 算力时，重组概率约 0.1%
  6 确认：攻击者有 10% 算力时，重组概率约 0.0001%（Satoshi 在白皮书中推荐）
 10 确认：攻击者有 10% 算力时，重组概率约 0.0000001%
```

### 2.2 Go 代码：算力攻击下重组概率模拟

```go
package main

import (
    "fmt"
    "math"
    "math/rand"
)

// ─── PoW 重组概率计算 ───

// ReorgProbabilityPoW 计算 PoW 链在给定确认数下被攻击者追上的概率
// 基于中本聪白皮书第 11 章的概率模型（简化版）
//
// q: 攻击者算力占比 (0 < q < 0.5)
// z: 确认区块数
//
// 返回攻击者成功追上的概率
func ReorgProbabilityPoW(q float64, z int) float64 {
    if q >= 0.5 {
        return 1.0 // 攻击者掌握 51%+ 算力，最终一定成功
    }

    p := 1.0 - q // 诚实节点算力占比

    // 泊松分布的期望值
    lambda := float64(z) * (q / p)

    // 中本聪公式：Σ P(Poisson=k) * (1 - (q/p)^(z-k))
    sum := 0.0
    for k := 0; k <= z; k++ {
        poisson := math.Pow(lambda, float64(k)) * math.Exp(-lambda) / factorial(k)
        prob := math.Pow(q/p, float64(z-k))
        if prob > 1.0 {
            prob = 1.0
        }
        sum += poisson * (1.0 - prob)
    }

    return sum
}

func factorial(n int) float64 {
    if n <= 1 {
        return 1
    }
    result := 1.0
    for i := 2; i <= n; i++ {
        result *= float64(i)
    }
    return result
}

// ─── PoW 链模拟器 ───

// Block 区块结构
type Block struct {
    Height       int64
    Hash         string
    ParentHash   string
    Miner        string // 矿工标识
    Transactions []string
}

// PoWChain PoW 区块链
type PoWChain struct {
    BlocksByHeight map[int64][]*Block // 同一高度可能有多个竞争区块（分叉）
    Tip            *Block             // 当前最长链的顶端
    TotalHeight    int64
}

// AddBlock 添加新区块，可能触发重组
func (c *PoWChain) AddBlock(block *Block) *ReorgEvent {
    c.BlocksByHeight[block.Height] = append(c.BlocksByHeight[block.Height], block)

    // 检查这个新区块是否导致链的 tip 变化
    if block.Height > c.TotalHeight {
        // 扩展当前最长链
        c.TotalHeight = block.Height
        c.Tip = block
        return nil // 正常扩展，无重组
    }

    // 可能产生分叉
    if block.Height == c.TotalHeight && block.Hash != c.Tip.Hash {
        return c.checkForReorg(block)
    }

    return nil
}

// ReorgEvent 重组事件
type ReorgEvent struct {
    Depth        int64    // 重组深度（废弃了几个区块）
    OrphanBlocks []*Block // 被废弃的区块
    NewBlocks    []*Block // 新加入的区块
    RevertedTXs  []string // 被回滚的交易
}

// checkForReorg 检查是否发生重组
func (c *PoWChain) checkForReorg(newBlock *Block) *ReorgEvent {
    // 在同一高度有两个不同区块，跟踪两条链的长度
    currentTip := c.Tip
    newTip := newBlock

    // 模拟：新链多出一条证明（实际中需要比较两链总工作量）
    // 这里简化为：新链拉出了更多后续区块就触发重组
    orphanHeight := newBlock.Height

    // 检查是否有区块在快照高度之上
    orphanBlocks := c.getBlocksAbove(orphanHeight - 1)

    event := &ReorgEvent{
        Depth:        int64(len(orphanBlocks)),
        OrphanBlocks: orphanBlocks,
        RevertedTXs:  collectTransactions(orphanBlocks),
    }

    // 切换 tip
    c.Tip = newBlock
    _ = currentTip // 被替代

    return event
}

func (c *PoWChain) getBlocksAbove(height int64) []*Block {
    var blocks []*Block
    for h := height + 1; h <= c.TotalHeight; h++ {
        if blks, ok := c.BlocksByHeight[h]; ok && len(blks) > 0 {
            blocks = append(blocks, blks[0])
        }
    }
    return blocks
}

func collectTransactions(blocks []*Block) []string {
    var txs []string
    for _, b := range blocks {
        txs = append(txs, b.Transactions...)
    }
    return txs
}

// ─── PoW 重组演示 ───

func demoPoWReorg() {
    fmt.Println("=== PoW 区块重组演示 ===")
    fmt.Println()

    chain := &PoWChain{
        BlocksByHeight: make(map[int64][]*Block),
    }

    // 正常出块 ─────────────────────────────────
    fmt.Println("【阶段1 - 正常出块】")
    for h := int64(1); h <= 5; h++ {
        block := &Block{
            Height:       h,
            Hash:         fmt.Sprintf("0xBLOCK_%d_A", h),
            ParentHash:   fmt.Sprintf("0xBLOCK_%d", h-1),
            Transactions: []string{fmt.Sprintf("TX_%d", h)},
        }
        chain.AddBlock(block)
        fmt.Printf("  区块 %d 已确认 → 交易 TX_%d 已到账\n", h, h)
    }
    fmt.Printf("当前链顶端: %s (高度 %d)\n", chain.Tip.Hash, chain.TotalHeight)

    // 分叉攻击 ─────────────────────────────────
    fmt.Println()
    fmt.Println("【阶段2 - 分叉出现（攻击者挖出竞争链）】")
    fmt.Println("攻击者从区块3开始挖了一条更长的竞争链...")
    for h := int64(4); h <= 7; h++ {
        forkBlock := &Block{
            Height:       h,
            Hash:         fmt.Sprintf("0xBLOCK_%d_ATTACK", h),
            ParentHash:   fmt.Sprintf("0xBLOCK_%d_ATTACK", h-1),
            Miner:        "attacker",
            Transactions: []string{fmt.Sprintf("ATTACK_TX_%d", h)},
        }
        event := chain.AddBlock(forkBlock)
        if event != nil {
            fmt.Println()
            fmt.Printf("  ⚠️ 发生 %d 块深度重组！\n", event.Depth)
            fmt.Printf("  被废弃的区块: ")
            for _, b := range event.OrphanBlocks {
                fmt.Printf("%s ", b.Hash)
            }
            fmt.Println()
            fmt.Printf("  被回滚的交易: %v\n", event.RevertedTXs)
        }
    }

    fmt.Println()
    fmt.Printf("重组后链顶端: %s (高度 %d)\n", chain.Tip.Hash, chain.TotalHeight)

    // 概率分析 ─────────────────────────────────
    fmt.Println()
    fmt.Println("【阶段3 - 重组概率分析】")
    attackerHashrates := []float64{0.05, 0.10, 0.20, 0.30}
    confirmations := []int{1, 2, 3, 4, 5, 6, 10, 12, 20}

    fmt.Println()
    fmt.Printf("%-12s", "确认数\\算力")
    for _, q := range attackerHashrates {
        fmt.Printf("%-14s", fmt.Sprintf("q=%.0f%%", q*100))
    }
    fmt.Println()
    fmt.Println("────────────────────────────────────────────────────────────────────")

    for _, z := range confirmations {
        fmt.Printf("z=%d          ", z)
        for _, q := range attackerHashrates {
            prob := ReorgProbabilityPoW(q, z)
            fmt.Printf("%-14s", fmt.Sprintf("%.8f", prob))
        }
        fmt.Println()
    }

    fmt.Println()
    fmt.Println("结论：6 确认后，即使攻击者有 10% 算力，重组概率已低于 0.2%")
    fmt.Println("      但永远不等于 0 —— 这就是 PoW 的'概率性最终确定'")
}
```

## 三、PoS 链的最终确定性（Ethereum Gasper 共识）

### 3.1 Gasper 共识机制

以太坊从 PoW 转向 PoS 后，引入了 **Gasper** 共识（Casper FFG + LMD GHOST），实现了真正意义上的"最终确定性"（finality）。

```
Epoch = 32 个 Slot（每个 Slot 12 秒，一个 Epoch 约 6.4 分钟）

Slot 0     Slot 1     ...    Slot 31    │  Slot 32    ...   Slot 63
│─────────│─────────│ ... │──────────│  │  │─────────│ ... │──────────│
│         │         │     │          │  │  │         │     │          │
└────── Epoch N ─────────────────────┘  │  └────── Epoch N+1 ──────────┘
         ↑                               │           ↑
    Epoch 边界投票                   上个 Epoch 边界被投票后 → justified
                                                        after 2 Epochs → finalized
```

### 3.2 状态转换：Proposed → Justified → Finalized

```
区块状态机：

  Proposed ──→ Justified ──→ Finalized
  (已提议)      (已辩护)      (已最终确定，不可逆)
                  ↑                ↑
          需要 2/3 验证者投票   需要连续两个 justified epoch
```

- **Proposed**：区块被提议但尚未获得足够投票
- **Justified**：获得 2/3 验证者投票，确认区块是合法的
- **Finalized**：在前一个 justified epoch 的基础上，下一个 epoch 也被 justified → 不可逆

### 3.3 Go 代码：Gasper 共识简化模拟

```go
package main

import (
    "fmt"
    "math/big"
    "sort"
)

// ─── Gasper 共识简化模拟 ───

// BlockState 区块状态
type BlockState int

const (
    StateProposed  BlockState = iota // 已提议
    StateJustified                   // 已辩护
    StateFinalized                   // 已最终确定（不可逆）
)

func (s BlockState) String() string {
    switch s {
    case StateProposed:
        return "Proposed"
    case StateJustified:
        return "Justified"
    case StateFinalized:
        return "Finalized ✓"
    default:
        return "Unknown"
    }
}

// Validator 验证者
type Validator struct {
    ID            int
    Stake         *big.Int // 质押金额
    IsHonest      bool     // 是否诚实节点
    CurrentVote   int64    // 当前Epoch投票的目标Epoch
    PreviousVote  int64    // 上一Epoch投票的目标Epoch
}

// GasperChain 实现了 Gasper 共识的链
type GasperChain struct {
    Blocks      map[int64]*PoSBlock  // slot → block
    Validators  []*Validator
    EpochSize   int64 // 每个 Epoch 的 Slot 数 (实际 ETH = 32)
    TotalStake  *big.Int
}

// PoSBlock PoS 区块
type PoSBlock struct {
    Slot       int64
    Epoch      int64
    State      BlockState
    Hash       string
    ParentHash string
    ProposerID int
    Timestamp  int64
}

// EpochBoundary Epoch 边界投票结果
type EpochBoundary struct {
    Epoch          int64
    TargetEpoch    int64  // FFG 投票的目标 Epoch
    VotesReceived  *big.Int
    TotalStake     *big.Int
    ThresholdMet   bool   // 是否达到 2/3
}

// ProcessEpochVote 处理 Epoch 边界的 FFG 投票
func (g *GasperChain) ProcessEpochVote(currentEpoch int64, targetEpoch int64) *EpochBoundary {
    boundary := &EpochBoundary{
        Epoch:         currentEpoch,
        TargetEpoch:   targetEpoch,
        VotesReceived: big.NewInt(0),
        TotalStake:    g.TotalStake,
    }

    // 统计投票
    for _, v := range g.Validators {
        if v.IsHonest {
            boundary.VotesReceived.Add(boundary.VotesReceived, v.Stake)
        }
    }

    // 检查是否达到 2/3 多数
    twoThirds := new(big.Int).Mul(g.TotalStake, big.NewInt(2))
    twoThirds.Div(twoThirds, big.NewInt(3))

    boundary.ThresholdMet = boundary.VotesReceived.Cmp(twoThirds) >= 0

    return boundary
}

// TryFinalize 尝试对区块进行最终确定
//
// Gasper 规则：
// 当 Epoch N 已被 justified，且 Epoch N+1 也被 justified（指向 N），
// 则 Epoch N 的所有区块被 finalized
func (g *GasperChain) TryFinalize() []int64 {
    var finalizedEpochs []int64
    epochs := g.getAllEpochs()
    sort.Slice(epochs, func(i, j int) bool { return epochs[i] < epochs[j] })

    for i := 0; i < len(epochs)-1; i++ {
        epochN := epochs[i]
        epochN1 := epochs[i+1]

        // 检查 Epoch N 是否已是 justified
        if !g.isEpochJustified(epochN) {
            continue
        }

        // 检查 Epoch N+1 是否已将投票指向 Epoch N
        if !g.isEpochJustifiedVotingFor(epochN1, epochN) {
            continue
        }

        // Epoch N 达到 finalized（不可逆）！
        g.setEpochFinalized(epochN)
        finalizedEpochs = append(finalizedEpochs, epochN)
    }

    return finalizedEpochs
}

func (g *GasperChain) isEpochJustified(epoch int64) bool {
    for _, block := range g.Blocks {
        if block.Epoch == epoch && block.State >= StateJustified {
            return true
        }
    }
    return false
}

func (g *GasperChain) isEpochJustifiedVotingFor(currentEpoch, targetEpoch int64) bool {
    // 检查当前 Epoch 的验证者是否投票支持 targetEpoch
    boundary := g.ProcessEpochVote(currentEpoch, targetEpoch)
    return boundary.ThresholdMet
}

func (g *GasperChain) setEpochFinalized(epoch int64) {
    for _, block := range g.Blocks {
        if block.Epoch == epoch {
            block.State = StateFinalized
        }
    }
}

func (g *GasperChain) getAllEpochs() []int64 {
    seen := make(map[int64]bool)
    for _, block := range g.Blocks {
        seen[block.Epoch] = true
    }
    var epochs []int64
    for e := range seen {
        epochs = append(epochs, e)
    }
    return epochs
}

// IsTransactionFinal 检查交易所在的区块是否已 finalized
func (g *GasperChain) IsTransactionFinal(slot int64) (bool, BlockState) {
    block, ok := g.Blocks[slot]
    if !ok {
        return false, StateProposed
    }
    return block.State == StateFinalized, block.State
}

// ─── PoS 重组演示 ───

func demoPoSReorg() {
    fmt.Println("=== PoS (Gasper) 共识重组演示 ===")
    fmt.Println()

    chain := &GasperChain{
        Blocks:      make(map[int64]*PoSBlock),
        EpochSize:   8, // 简化为 8 个 Slot/Epoch，方便演示
        TotalStake:  big.NewInt(1000000),
    }

    // 创建验证者（大多数诚实）
    for i := 0; i < 100; i++ {
        chain.Validators = append(chain.Validators, &Validator{
            ID:       i,
            Stake:    big.NewInt(10000),
            IsHonest: i < 90, // 90% 诚实
        })
    }

    // 模拟 Epoch 0 出块并投票
    fmt.Println("【Epoch 0】")
    for slot := int64(0); slot < chain.EpochSize; slot++ {
        block := &PoSBlock{
            Slot:       slot,
            Epoch:      0,
            State:      StateProposed,
            Hash:       fmt.Sprintf("0xSLOT_%d", slot),
            ProposerID: int(slot) % 100,
        }
        chain.Blocks[slot] = block
    }

    // Epoch 0 结束时投票（Checkpoint）
    vote0 := chain.ProcessEpochVote(0, 0)
    fmt.Printf("  Epoch 0 投票结果: 得票 %s / 总质押 %s (需要 2/3 = %d%%)\n",
        vote0.VotesReceived, vote0.TotalStake, 67)
    if vote0.ThresholdMet {
        fmt.Println("  ✓ 投票达到 2/3 → Epoch 0 被 Justified")
        for _, b := range chain.Blocks {
            if b.Epoch == 0 {
                b.State = StateJustified
            }
        }
    }

    // 模拟 Epoch 1 出块并投票指向 Epoch 0
    fmt.Println()
    fmt.Println("【Epoch 1】（投票目标 → Epoch 0）")
    for slot := chain.EpochSize; slot < chain.EpochSize*2; slot++ {
        block := &PoSBlock{
            Slot:       slot,
            Epoch:      1,
            State:      StateProposed,
            Hash:       fmt.Sprintf("0xSLOT_%d", slot),
            ProposerID: int(slot) % 100,
        }
        chain.Blocks[slot] = block
    }
    vote1 := chain.ProcessEpochVote(1, 0) // 投票指向 Epoch 0
    fmt.Printf("  Epoch 1 投票结果: 得票 %s，目标 Epoch 0\n", vote1.VotesReceived)
    if vote1.ThresholdMet {
        fmt.Println("  ✓ 投票达到 2/3 → Epoch 1 被 Justified")
        for _, b := range chain.Blocks {
            if b.Epoch == 1 {
                b.State = StateJustified
            }
        }
    }

    // 尝试 Finalize
    fmt.Println()
    finalized := chain.TryFinalize()
    if len(finalized) > 0 {
        fmt.Printf("  🔒 Epoch %v 已 FINALIZED（不可逆！）\n", finalized)
    }

    // 模拟 Epoch 2 出块并投票 → Epoch 1 finalized
    fmt.Println()
    fmt.Println("【Epoch 2】（投票目标 → Epoch 1）")
    for slot := chain.EpochSize * 2; slot < chain.EpochSize*3; slot++ {
        block := &PoSBlock{
            Slot:       slot,
            Epoch:      2,
            State:      StateProposed,
            Hash:       fmt.Sprintf("0xSLOT_%d", slot),
            ProposerID: int(slot) % 100,
        }
        chain.Blocks[slot] = block
    }
    vote2 := chain.ProcessEpochVote(2, 1)
    if vote2.ThresholdMet {
        for _, b := range chain.Blocks {
            if b.Epoch == 2 {
                b.State = StateJustified
            }
        }
    }
    finalized2 := chain.TryFinalize()
    if len(finalized2) > 0 {
        fmt.Printf("  🔒 Epoch %v 已 FINALIZED（不可逆！）\n", finalized2)
    }

    // 总结 ─────────────────────────────────
    fmt.Println()
    fmt.Println("【状态总结】")
    for slot := int64(0); slot < chain.EpochSize*3; slot += 4 {
        b := chain.Blocks[slot]
        final, _ := chain.IsTransactionFinal(slot)
        fmt.Printf("  Slot %3d (Epoch %d): %-14s 已最终确定? %v\n",
            b.Slot, b.Epoch, b.State, final)
    }

    fmt.Println()
    fmt.Println("关键结论：")
    fmt.Println("  • Epoch 边界投票达到 2/3 → Justified")
    fmt.Println("  • 连续两个 Epoch 都被 Justified → 前一个 Epoch Finalized")
    fmt.Println("  • Finalized 后 → 数学上不可逆（除非 >1/3 验证者被罚没）")
    fmt.Println("  • 从提议到 Finalized 约需 2 个 Epoch ≈ 12.8 分钟")
}
```

## 四、交易所防御策略

### 4.1 核心防御原则

```
原则：等足够确认数才入账，金额越大等越久

小额充值（< $10K）  →  最小确认数
中额充值（$10K-$100K） →  2x 最小确认数
大额充值（>$100K）  →  3x 最小确认数 + 人工审核
机构充值（>$1M）    →  finalized 后 + 人工审核
```

```go
// AmountRange 金额区间
type AmountRange struct {
    Min          *big.Float
    Max          *big.Float
    Multiplier   float64 // 确认数倍率
    RequireManual bool   // 是否需要人工审核
    Description  string
}

// 金额分层配置
var AmountTiers = []AmountRange{
    {Min: big.NewFloat(0), Max: big.NewFloat(10000), Multiplier: 1.0, RequireManual: false, Description: "小额"},
    {Min: big.NewFloat(10000), Max: big.NewFloat(100000), Multiplier: 2.0, RequireManual: false, Description: "中额"},
    {Min: big.NewFloat(100000), Max: big.NewFloat(1000000), Multiplier: 3.0, RequireManual: true, Description: "大额"},
    {Min: big.NewFloat(1000000), Max: nil, Multiplier: 5.0, RequireManual: true, Description: "机构级"},
}

// GetRequiredConfirmations 根据金额和链计算所需确认数
func GetRequiredConfirmations(chain string, amountUSD *big.Float) (int64, bool, string) {
    config, ok := ChainConfigs[chain]
    if !ok {
        return 999, true, "未知链，默认无限确认+人工审核"
    }

    // 遍历金额分层，找到匹配的区间
    for _, tier := range AmountTiers {
        if tier.Max == nil || amountUSD.Cmp(tier.Max) <= 0 {
            requiredConfs := int64(float64(config.MinConfirmations) * tier.Multiplier)
            return requiredConfs, tier.RequireManual, tier.Description
        }
    }

    return config.MinConfirmations, false, "默认"
}

// DepositStatus 充值状态
type DepositStatus int

const (
    StatusPending    DepositStatus = iota // 等待确认
    StatusConfirming                      // 确认中（确认数不够）
    StatusConfirmed                       // 已确认（可入账）
    StatusManualReview                    // 人工审核中
    StatusCredited                        // 已入账
    StatusReorged                         // ⚠️ 已重组（回滚）
    StatusRejected                        // 已拒绝
)

func (s DepositStatus) String() string {
    return [...]string{
        "等待确认", "确认中", "已确认", "人工审核中", "已入账", "⚠️已重组", "已拒绝",
    }[s]
}

// Deposit 充值交易
type Deposit struct {
    TXID         string
    Chain        string
    FromAddress  string
    ToAddress    string
    Amount       *big.Float     // 代币数量
    AmountUSD    *big.Float     // 美元价值
    Asset        string         // 资产类型

    FirstSeenBlock     int64     // 首次出现的区块高度
    CurrentBlock       int64     // 当前链上区块高度
    Confirmations      int64     // 当前确认数
    RequiredConfs      int64     // 所需确认数
    Status             DepositStatus
    ManualReviewNeeded bool

    FirstSeenAt        time.Time
    LastCheckedAt      time.Time
    CreditedAt         time.Time

    mu sync.RWMutex
}

// DepositTracker 充值追踪器
type DepositTracker struct {
    deposits map[string]*Deposit // txid → deposit
    mu       sync.RWMutex
}

// CheckConfirmations 检查确认数并决定是否入账
func (dt *DepositTracker) CheckConfirmations(deposit *Deposit, currentChainHeight int64) {
    deposit.mu.Lock()
    defer deposit.mu.Unlock()

    deposit.CurrentBlock = currentChainHeight
    deposit.Confirmations = currentChainHeight - deposit.FirstSeenBlock + 1
    deposit.LastCheckedAt = time.Now()

    // 检查是否发生重组
    if deposit.Confirmations < 0 {
        fmt.Printf("  ⚠️ %s: 充值交易可能已重组！当前确认数=%d\n",
            deposit.TXID, deposit.Confirmations)
        deposit.Status = StatusReorged
        return
    }

    // 判断是否可以入账
    if deposit.Status >= StatusCredited {
        return // 已入账，不需要再处理
    }

    if deposit.ManualReviewNeeded && deposit.Status < StatusManualReview {
        deposit.Status = StatusManualReview
        fmt.Printf("  🔔 %s: 大额充值(%s USD) 需要人工审核\n",
            deposit.TXID, formatUSD(deposit.AmountUSD))
        return
    }

    if deposit.Confirmations >= deposit.RequiredConfs &&
        deposit.Status == StatusManualReview {
        // 人工审核已通过 → 入账
        deposit.Status = StatusCredited
        deposit.CreditedAt = time.Now()
        fmt.Printf("  ✓ %s: 人工审核通过，已入账 %s %s\n",
            deposit.TXID, formatAmount(deposit.Amount), deposit.Asset)
        return
    }

    if deposit.Confirmations >= deposit.RequiredConfs &&
        deposit.Status < StatusConfirmed {
        deposit.Status = StatusConfirmed
    }
}

// NewDeposit 创建新的充值记录
func (dt *DepositTracker) NewDeposit(
    txID, chain, asset string,
    amount, amountUSD *big.Float,
    blockHeight int64,
) *Deposit {

    config := ChainConfigs[chain]
    requiredConfs, needManual, tier := GetRequiredConfirmations(chain, amountUSD)

    deposit := &Deposit{
        TXID:               txID,
        Chain:              chain,
        Asset:              asset,
        Amount:             amount,
        AmountUSD:          amountUSD,
        FirstSeenBlock:     blockHeight,
        CurrentBlock:       blockHeight,
        RequiredConfs:      requiredConfs,
        ManualReviewNeeded: needManual,
        Status:             StatusPending,
        FirstSeenAt:        time.Now(),
    }

    dt.mu.Lock()
    dt.deposits[txID] = deposit
    dt.mu.Unlock()

    fmt.Printf("📥 新充值: %s | %s %s | 约$%s | %s链 | 需要%d确认(%s) | 金额等级:%s\n",
        txID[:16]+"...",
        formatAmount(amount), asset,
        formatUSD(amountUSD),
        chain,
        requiredConfs,
        config.FinalityModel,
        tier,
    )

    if needManual {
        fmt.Printf("  ⚠️ 金额等级为 %s，达标后需人工审核\n", tier)
    }

    return deposit
}

// ReorgDetector 重组检测器
type ReorgDetector struct {
    LastSeenHash map[int64]string // height → block_hash
    mu           sync.RWMutex
}

// CheckForReorg 检查是否发生重组
// 如果同一高度的区块哈希变了，说明发生了重组
func (rd *ReorgDetector) CheckForReorg(height int64, newHash string) *ReorgEvent {
    rd.mu.RLock()
    oldHash, exists := rd.LastSeenHash[height]
    rd.mu.RUnlock()

    if exists && oldHash != newHash {
        // 发生重组！
        rd.mu.Lock()
        rd.LastSeenHash[height] = newHash
        rd.mu.Unlock()

        return &ReorgEvent{
            Height:    height,
            OldHash:   oldHash,
            NewHash:   newHash,
            Timestamp: time.Now(),
        }
    }

    rd.mu.Lock()
    rd.LastSeenHash[height] = newHash
    rd.mu.Unlock()

    return nil
}

type ReorgEvent struct {
    Height    int64
    OldHash   string
    NewHash   string
    Timestamp time.Time
}

// HandleReorg 处理重组：标记受影响的所有充值交易
func (dt *DepositTracker) HandleReorg(reorgHeight int64) []string {
    var affected []string

    dt.mu.RLock()
    defer dt.mu.RUnlock()

    for txID, deposit := range dt.deposits {
        deposit.mu.RLock()
        if deposit.FirstSeenBlock >= reorgHeight && deposit.Status >= StatusCredited {
            // 这个充值交易可能被回滚了！
            affected = append(affected, txID)
        }
        deposit.mu.RUnlock()
    }

    if len(affected) > 0 {
        fmt.Printf("⚠️⚠️⚠️ 检测到高度 %d 发生重组，以下 %d 笔已入账交易受影响！\n",
            reorgHeight, len(affected))
    }

    return affected
}

        name   string
        config ChainConfig
        wait   time.Duration
    }
    var chains []chainInfo
    for name, cfg := range ChainConfigs {
        wait := time.Duration(cfg.MinConfirmations) * cfg.BlockTime
        chains = append(chains, chainInfo{name, cfg, wait})
    }
    sort.Slice(chains, func(i, j int) bool {
        return chains[i].wait < chains[j].wait
    })

    for _, c := range chains {
        wait := c.wait
        var waitStr string
        if wait < time.Minute {
            waitStr = fmt.Sprintf("%.1f 秒", wait.Seconds())
        } else if wait < time.Hour {
            waitStr = fmt.Sprintf("%.1f 分钟", wait.Minutes())
        } else {
            waitStr = fmt.Sprintf("%.1f 小时", wait.Hours())
        }
        fmt.Printf("%-8s %-16d %-14s %-16s %s\n",
            c.name,
            c.config.MinConfirmations,
            c.config.FinalityModel,
            c.config.BlockTime,
            waitStr,
        )
    }

    fmt.Println()
    fmt.Println("注：ETH PoS 最终确定性需约 2 Epoch（~12.8 分钟），12 区块仅为初步确认")
}
```

## 六、PoW vs PoS 重组对比

| 维度 | PoW 链（BTC、旧ETH、BSC） | PoS 链（新ETH） |
|------|--------------------------|----------------|
| 最终确定性 | **概率性**：永远可能重组 | **确定性**：finalized 后不可逆 |
| 安全模型 | 算力博弈（经济 + 物理） | 质押博弈（纯经济） |
| 重组概率 | 随确认增加指数下降 | finalized 后 = 0 |
| 大规模回滚 | 理论上可能（51% 攻击） | 需要 >1/3 验证者被罚没 |
| 防御策略 | 多等确认数 | 等 finalized 状态 |
| 恢复机制 | 最长链自动切换 | 罚没违规验证者 |
| 最长重组记录 | ETH PoW: ~7 块（2020/08） | ETH PoS: 极少发生 |

## 七、总结：交易所防御清单

```
□  1. 不同链配置不同确认阈值（BTC 6 / ETH 12+finalized / BSC 20 / Polygon 64）
□  2. 金额分层：大额充值需要更多确认数
□  3. 大额充值（>$100K）加人工审核环节
□  4. 实时监控链上区块哈希变化，一旦检测到重组立即告警
□  5. 对已入账但确认数不足的交易建立"可回滚"标记
□  6. 在桥接/L2 场景下，额外考虑 L1 的最终确定时间
□  7. 永远假设链可能重组，在设计上做到零容忍
```
